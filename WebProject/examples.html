أمثلة عملية على استخدام المشروع

📢 مثال 1: الإخطارات

استخدام الإخطارات لعرض الرسائل للمستخدم:

// إظهار إخطار نجاح
showNotification('تم إرسال الرسالة بنجاح', 'success');

// إظهار إخطار خطأ
showNotification('حدث خطأ ما', 'error');

// إظهار تحذير
showNotification('تأكد من المدخلات', 'warning');

// إظهار معلومة
showNotification('هذه معلومة مهمة', 'info');

🎬 مثال 2: الحركات والرسوم المتحركة

استخدام مكتبة الحركات المدمجة:

// ظهور تدريجي
Animations.fadeIn('.element');

// دوران
Animations.rotate('.element');

// قفز
Animations.bounce('.element');

// نبض
Animations.pulse('.element');

// اهتزاز
Animations.shake('.element');

🖼️ مثال 3: إدارة واجهة المستخدم

استخدام مكتبة UI لإدارة العناصر:

// إظهار/إخفاء عناصر
UI.show('.element');
UI.hide('.element');
UI.toggle('.element');

// تغيير النصوص
UI.setText('.element', 'نص جديد');

// إضافة/إزالة فئات
UI.addClass('.element', 'active');
UI.removeClass('.element', 'active');
هذا النص سيتغير

💾 مثال 4: تخزين البيانات محلياً

حفظ واسترجاع البيانات باستخدام localStorage:

// حفظ البيانات
saveData('userName', 'أحمد');
saveData('userAge', 25);

// استرجاع البيانات
let name = getData('userName');

// حذف البيانات
removeData('userName');

📝 مثال 5: معالجة النماذج

التعامل مع النماذج والتحقق من الصحة:

// التحقق من صحة النموذج
let isValid = UI.validateForm('#myForm');

// الحصول على بيانات النموذج
let data = UI.getFormData('#myForm');
console.log(data);

📦 مثال 6: نوافذ منفثقة (Modals)

عرض محتوى في نوافذ منفثقة:

let content = '<h3>مرحباً!</h3><p>هذا محتوى النافذة المنفثقة</p>';
UI.showModal(content, 'العنوان');

🌍 مثال 7: تأثير Parallax

حركة الخلفية ببطء أثناء التمرير:

Animations.parallax('.parallax-element', 0.5);
📍 Parallax Element

📚 ملخص الدوال الرئيسية

الدالة الوصف
showNotification() عرض إخطار للمستخدم
saveData() حفظ البيانات في localStorage
getData() استرجاع البيانات من localStorage
log() تسجيل الرسائل في console
UI.show() / hide() إظهار/إخفاء العناصر
Animations.* مكتبة الحركات والرسوم المتحركة
